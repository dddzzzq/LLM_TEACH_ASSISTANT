import * as https from 'https';
import * as http from 'http';
import * as fs from 'fs';
import * as path from 'path';
import { URL } from 'url';

interface DownloadProgress {
    downloadedBytes: number;
    totalBytes: number;
    startTime: number;
}

class FileDownloader {
    private progress: DownloadProgress = {
        downloadedBytes: 0,
        totalBytes: 0,
        startTime: Date.now()
    };

    private progressInterval?: NodeJS.Timeout;

    constructor(private url: string, private outputPath: string) {}

    async download(): Promise<void> {
        return new Promise((resolve, reject) => {
            const urlObj = new URL(this.url);
            const protocol = urlObj.protocol === 'https:' ? https : http;

            // 确保输出目录存在
            const outputDir = path.dirname(this.outputPath);
            if (!fs.existsSync(outputDir)) {
                fs.mkdirSync(outputDir, { recursive: true });
            }

            const request = protocol.get(this.url, (response) => {
                // 检查响应状态
                if (response.statusCode !== 200) {
                    reject(new Error(`下载失败: HTTP ${response.statusCode}`));
                    return;
                }

                // 获取文件总大小
                const contentLength = response.headers['content-length'];
                this.progress.totalBytes = contentLength ? parseInt(contentLength) : 0;

                // 创建写入流
                const fileStream = fs.createWriteStream(this.outputPath);
                
                // 开始进度监控
                this.startProgressMonitoring();

                // 处理数据流
                response.pipe(fileStream);

                response.on('data', (chunk: Buffer) => {
                    this.progress.downloadedBytes += chunk.length;
                });

                response.on('end', () => {
                    this.stopProgressMonitoring();
                    fileStream.close();
                    this.printFinalProgress();
                    resolve();
                });

                response.on('error', (error) => {
                    this.stopProgressMonitoring();
                    reject(error);
                });

                fileStream.on('error', (error) => {
                    this.stopProgressMonitoring();
                    reject(error);
                });
            });

            request.on('error', (error) => {
                this.stopProgressMonitoring();
                reject(error);
            });
        });
    }

    private startProgressMonitoring(): void {
        this.progressInterval = setInterval(() => {
            this.printProgress();
        }, 200);
    }

    private stopProgressMonitoring(): void {
        if (this.progressInterval) {
            clearInterval(this.progressInterval);
            this.progressInterval = undefined;
        }
    }

    private printProgress(): void {
        const elapsedSeconds = (Date.now() - this.progress.startTime) / 1000;
        const minutes = Math.floor(elapsedSeconds / 60);
        const seconds = Math.floor(elapsedSeconds % 60);
        const downloadedKB = (this.progress.downloadedBytes / 1024).toFixed(2);

        let progressText = `已下载：${downloadedKB} KB，已用时：${minutes}分${seconds}秒`;

        if (this.progress.totalBytes > 0) {
            const percentage = ((this.progress.downloadedBytes / this.progress.totalBytes) * 100).toFixed(1);
            const totalKB = (this.progress.totalBytes / 1024).toFixed(2);
            progressText += `，进度：${percentage}% (${downloadedKB}/${totalKB} KB)`;
        }

        // 使用回车符覆盖上一行
        process.stdout.write('\r' + progressText);
    }

    private printFinalProgress(): void {
        const elapsedSeconds = (Date.now() - this.progress.startTime) / 1000;
        const minutes = Math.floor(elapsedSeconds / 60);
        const seconds = Math.floor(elapsedSeconds % 60);
        const downloadedKB = (this.progress.downloadedBytes / 1024).toFixed(2);

        console.log(`\n下载完成！文件已保存到 ${this.outputPath}`);
        console.log(`总计下载：${downloadedKB} KB，总用时：${minutes}分${seconds}秒`);
    }
}

// 命令行参数处理
function main(): void {
    const args = process.argv.slice(2);

    if (args.length !== 2) {
        console.log('使用方法: filedownloader <url> <outputfile>');
        console.log('示例: filedownloader https://example.com/sample.txt ./sample.txt');
        process.exit(1);
    }

    const [url, outputPath] = args;

    // 验证 URL
    try {
        new URL(url);
    } catch (error) {
        console.error('错误: 无效的 URL');
        process.exit(1);
    }

    const downloader = new FileDownloader(url, outputPath);

    downloader.download()
        .then(() => {
            process.exit(0);
        })
        .catch((error) => {
            console.error(`\n下载失败: ${error.message}`);
            process.exit(1);
        });
}

// 启动程序
if (require.main === module) {
    main();
}

export { FileDownloader };

function formatTimeDuration(ms: number): string {
  const totalSeconds = Math.floor(ms / 1000);
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = totalSeconds % 60;
  return `${minutes}分${seconds}秒`;

function humanKB(bytes: number): string {
  return (bytes / 1024).toFixed(2);
}

/**
 * 发起 GET 请求，返回 response 流（并处理重定向）。
 * 支持最多 redirectLimit 次重定向。
 */
function fetchStream(targetUrl: string, redirectLimit = 10): Promise<http.IncomingMessage> {
  return new Promise((resolve, reject) => {
    if (redirectLimit < 0) {
      reject(new Error("Too many redirects"));
      return;
    }

    const parsed = url.parse(targetUrl);
    const lib: RequestModule = parsed.protocol === "https:" ? https : http;

    const req = lib.get(targetUrl, (res) => {
      const status = res.statusCode ?? 0;
      // 处理重定向（3xx）
      if (status >= 300 && status < 400 && res.headers.location) {
        const location = url.resolve(targetUrl, res.headers.location);
        // consume and discard current response
        res.resume();
        fetchStream(location, redirectLimit - 1).then(resolve, reject);
        return;
      }

      if (status >= 200 && status < 300) {
        resolve(res);
      } else {
        // 非成功状态码，收集错误信息
        let body = "";
        res.on("data", (chunk) => (body += chunk.toString()));
        res.on("end", () => {
          reject(new Error(`Request failed with status ${status}. Body: ${body}`));
        });
      }
    });

    req.on("error", (err) => reject(err));
    // 超时处理（可选）：10s 链接超时
    req.setTimeout(10000, () => {
      req.abort();
      reject(new Error("Request timeout"));
    });
  });
}

async function downloadToFile(targetUrl: string, outPath: string): Promise<void> {
  // 验证目录可写
  const dir = require("path").dirname(outPath);
  try {
    fs.mkdirSync(dir, { recursive: true });
  } catch (err) {

  }

  const startTime = Date.now();
  const response = await fetchStream(targetUrl);

  const totalBytesHeader = response.headers["content-length"];
  const totalBytes = totalBytesHeader ? parseInt(totalBytesHeader, 10) : NaN;

  const fileStream = fs.createWriteStream(outPath);
  let downloaded = 0;

  // 每 200ms 输出一次进度
  const intervalMs = 200;
  const interval = setInterval(() => {
    const elapsed = Date.now() - startTime;
    const kb = humanKB(downloaded);
    const timeStr = formatTimeDuration(elapsed);
    if (!isNaN(totalBytes)) {
      const pct = ((downloaded / totalBytes) * 100).toFixed(2);
      process.stdout.write(`已下载: ${kb} KB (${pct}%)，已用时: ${timeStr}\r`);
    } else {
      process.stdout.write(`已下载: ${kb} KB，已用时: ${timeStr}\r`);
    }
  }, intervalMs);

  // 错误处理：response 或 fileStream 上的错误
  const cleanupAndReject = (err: Error) => {
    clearInterval(interval);
    // 尝试关闭流
    try { fileStream.close(); } catch {}
    try { response.destroy(); } catch {}
    throw err;
  };

  return new Promise<void>((resolve, reject) => {
    response.on("data", (chunk: Buffer) => {
      downloaded += chunk.length;
    });

    response.on("error", (err) => {
      clearInterval(interval);
      reject(err);
    });

    fileStream.on("error", (err) => {
      clearInterval(interval);
      reject(err);
    });

    response.pipe(fileStream);

    fileStream.on("finish", () => {
      clearInterval(interval);
      // 最终一行换行并输出完成信息
      const elapsed = Date.now() - startTime;
      const timeStr = formatTimeDuration(elapsed);
      const kb = humanKB(downloaded);
      // 打印最终状态（换行）
      process.stdout.write(`已下载: ${kb} KB，已用时: ${timeStr}\n`);
      console.log(`下载完成!文件已保存到 ${outPath}`);
      resolve();
    });

    // 如果连接中断
    response.on("close", () => {
        
    });
  });
}

import * as https from 'https';
import * as http from 'http';
import * as fs from 'fs';
import * as path from 'path';
import { URL } from 'url';

/**
 * 格式化时间为"分秒"格式
 */
function formatTime(seconds: number): string {
  const minutes = Math.floor(seconds / 60);
  const secs = Math.floor(seconds % 60);
  return `${minutes}分${secs}秒`;
}

/**
 * 格式化文件大小为KB
 */
function formatSize(bytes: number): string {
  return (bytes / 1024).toFixed(2);
}

/**
 * 下载文件
 * @param url 下载URL
 * @param outputPath 保存路径
 */
export async function downloadFile(url: string, outputPath: string): Promise<void> {
  return new Promise((resolve, reject) => {
    try {
      const parsedUrl = new URL(url);
      const protocol = parsedUrl.protocol === 'https:' ? https : http;
      
      // 确保输出目录存在
      const outputDir = path.dirname(outputPath);
      if (!fs.existsSync(outputDir)) {
        fs.mkdirSync(outputDir, { recursive: true });
      }

      // 创建写入流
      const fileStream = fs.createWriteStream(outputPath);
      
      // 下载统计
      let downloadedBytes = 0;
      const startTime = Date.now();
      
      // 进度显示定时器
      const progressInterval = setInterval(() => {
        const elapsedSeconds = (Date.now() - startTime) / 1000;
        console.log(`已下载: ${formatSize(downloadedBytes)}KB，已用时: ${formatTime(elapsedSeconds)}`);
      }, 200);

      // 发起HTTP/HTTPS请求
      const request = protocol.get(url, (response) => {
        // 处理重定向
        if (response.statusCode === 301 || response.statusCode === 302) {
          clearInterval(progressInterval);
          fileStream.close();
          fs.unlinkSync(outputPath);
          
          const redirectUrl = response.headers.location;
          if (redirectUrl) {
            downloadFile(redirectUrl, outputPath)
              .then(resolve)
              .catch(reject);
          } else {
            reject(new Error('重定向失败：未找到目标URL'));
          }
          return;
        }

        // 检查响应状态
        if (response.statusCode !== 200) {
          clearInterval(progressInterval);
          fileStream.close();
          fs.unlinkSync(outputPath);
          reject(new Error(`下载失败: HTTP ${response.statusCode}`));
          return;
        }

        // 接收数据
        response.on('data', (chunk: Buffer) => {
          downloadedBytes += chunk.length;
          fileStream.write(chunk);
        });

        // 下载完成
        response.on('end', () => {
          clearInterval(progressInterval);
          fileStream.end();
          
          // 最后一次显示进度
          const elapsedSeconds = (Date.now() - startTime) / 1000;
          console.log(`已下载: ${formatSize(downloadedBytes)}KB，已用时: ${formatTime(elapsedSeconds)}`);
          console.log(`\n下载完成! 文件已保存到 ${outputPath}`);
          
          resolve();
        });

        // 错误处理
        response.on('error', (err) => {
          clearInterval(progressInterval);
          fileStream.close();
          if (fs.existsSync(outputPath)) {
            fs.unlinkSync(outputPath);
          }
          reject(err);
        });
      });

      // 请求错误处理
      request.on('error', (err) => {
        clearInterval(progressInterval);
        fileStream.close();
        if (fs.existsSync(outputPath)) {
          fs.unlinkSync(outputPath);
        }
        reject(err);
      });

      // 文件写入错误处理
      fileStream.on('error', (err) => {
        clearInterval(progressInterval);
        request.destroy();
        if (fs.existsSync(outputPath)) {
          fs.unlinkSync(outputPath);
        }
        reject(err);
      });

    } catch (err) {
      reject(err);
    }
  });
}

async function downloadFile(url: string, outputPath: string) {
  const writer = fs.createWriteStream(outputPath);
  const response = await axios({
    method: 'GET',
    url,
    responseType: 'stream',
  });

  let downloaded = 0;
  let lastLogged = 0;
  const startTime = Date.now();

  response.data.on('data', (chunk: Buffer) => {
    downloaded += chunk.length;
  });

  response.data.pipe(writer);

  const timer = setInterval(() => {
    if (downloaded > lastLogged) {
      const kb = (downloaded / 1024).toFixed(2);
      const time = `${Date.now() - startTime}毫秒`;
      console.log(`已下载：${kb} KB，已用时：${time}`);
      lastLogged = downloaded;
    }
  }, 200);

  return new Promise<void>((resolve, reject) => {
    writer.on('finish', () => {
      clearInterval(timer);
      // 最后一次补刷，确保 100 % 显示
      const kb = (downloaded / 1024).toFixed(2);
